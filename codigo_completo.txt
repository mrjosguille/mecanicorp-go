const RutaProtegida = ({ children, rolesPermitidos }) => {
  const { session, perfil, loading } = useAuth();

  if (loading) return <div style={{ padding: '50px', textAlign: 'center' }}>Cargando sistema...</div>;
  if (!session) return <Navigate to="/login" />;
  
  if (rolesPermitidos && perfil && !rolesPermitidos.includes(perfil.rol)) {
    return (
      <div style={{ padding: '50px', textAlign: 'center', color: '#ef4444' }}>
        <h2>Acceso Denegado</h2>
        <p>No tienes permisos para ver esta sección.</p>
      </div>
    );
  }

  return children ? children : <Outlet />;
};

      const canvas = canvasRef.current;
      const ctx = canvas.getContext('2d');
      ctx.strokeStyle = '#000';
      ctx.lineWidth = 2;
      ctx.lineCap = 'round';

      const getCoordinates = (e) => {
        const rect = canvas.getBoundingClientRect();
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: clientX - rect.left, y: clientY - rect.top };
      };

      const startDrawing = (e) => { e.preventDefault(); setIsDrawing(true); const { x, y } = getCoordinates(e); ctx.beginPath(); ctx.moveTo(x, y); };
      const draw = (e) => { if (!isDrawing) return; e.preventDefault(); const { x, y } = getCoordinates(e); ctx.lineTo(x, y); ctx.stroke(); };
      const stopDrawing = () => { setIsDrawing(false); };

      canvas.addEventListener('mousedown', startDrawing); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', stopDrawing); canvas.addEventListener('mouseleave', stopDrawing);
      canvas.addEventListener('touchstart', startDrawing, { passive: false }); canvas.addEventListener('touchmove', draw, { passive: false }); canvas.addEventListener('touchend', stopDrawing);

      return () => {
        canvas.removeEventListener('mousedown', startDrawing); canvas.removeEventListener('mousemove', draw); canvas.removeEventListener('mouseup', stopDrawing); canvas.removeEventListener('mouseleave', stopDrawing);
        canvas.removeEventListener('touchstart', startDrawing); canvas.removeEventListener('touchmove', draw); canvas.removeEventListener('touchend', stopDrawing);

    setBusqueda(texto.toUpperCase());
    if (texto.trim() === '') { setResultadosBusqueda(todosLosPendientes); return; }
    const filtrados = todosLosPendientes.filter(r => r.placa.includes(texto.toUpperCase()) || r.numero_proforma?.toString().includes(texto) || r.descripcion.toUpperCase().includes(texto.toUpperCase()) || r.cliente.toUpperCase().includes(texto.toUpperCase()));
    setResultadosBusqueda(filtrados);

    }]);
    if (error) return alert("Error: " + error.message);

      }));
      setSolicitudes(completas);

  };

    if (actasData && actasData.length > 0) {

        <div style={{ background: 'white', padding: '20px', borderRadius: '8px' }}>

                      <div>
                          <strong>Acta #{acta.id.slice(0,6)}</strong> - Entregado a: <strong>{acta.detalles[0]?.empleado}</strong> - Fecha: {new Date(acta.fecha_entrega).toLocaleString()}
                      </div>

                  </div>
                  <div style={{ padding: '10px 15px' }}>
                      <ul style={{ margin: 0, paddingLeft: '20px', fontSize: '13px' }}>
                          {acta.detalles.map(d => (
                              <li key={d.id}>{d.placa} - {d.cantidad_solicitada}x {d.nombreMaterial}</li>
                          ))}
                      </ul>
                  </div>
              </div>
          ))}
          {historialEntregasFiltrado.length === 0 && <p>No se encontraron entregas.</p>}
        </div>
      )}

      {/* MODAL DE RECEPCIÓN */}
      {repuestoSeleccionado && (
        <div style={{ position: 'fixed', top: 0, left: 0, width: '100vw', height: '100vh', background: 'rgba(0,0,0,0.5)', display: 'flex', justifyContent: 'center', alignItems: 'center', zIndex: 1000 }}>
          <div style={{ background: 'white', padding: '30px', borderRadius: '8px', width: '450px' }}>
            <h3 style={{ marginBottom: '15px', color: '#1e4b8f' }}>Revisión de Llegada</h3>

            <div style={{ display: 'flex', justifyContent: 'flex-end', gap: '10px' }}>
              <button onClick={() => setRepuestoSeleccionado(null)} style={{ padding: '10px 20px', cursor: 'pointer' }}>Cancelar</button>
              <button onClick={guardarRecepcion} style={{ padding: '10px 20px', background: '#27ae60', color: 'white', border: 'none', borderRadius: '5px', cursor: 'pointer' }}>Guardar</button>
            </div>
          </div>
        </div>
      )}

      {/* MODAL DE ENTREGA MÚLTIPLE CON FIRMA */}
      {modalEntregaAbierto && (
        <div style={{ position: 'fixed', top: 0, left: 0, width: '100vw', height: '100vh', background: 'rgba(0,0,0,0.5)', display: 'flex', justifyContent: 'center', alignItems: 'center', zIndex: 1000 }}>
          <div style={{ background: 'white', padding: '30px', borderRadius: '8px', width: '500px' }}>
            <h3 style={{ marginBottom: '10px', color: '#1e4b8f' }}>Acta de Entrega Múltiple</h3>
            <p style={{ fontSize: '14px', marginBottom: '20px' }}>
              Entregando <strong>{seleccionadas.length} materiales</strong> a <strong>{solicitudes.find(s => s.id === seleccionadas[0])?.ordenInfo?.empleados?.nombre}</strong>.
            </p>
            
            <label style={{ display: 'block', marginBottom: '10px', fontWeight: 'bold' }}>Firma del Chapista (Receptor):</label>
            <div style={{ border: '2px dashed #ccc', borderRadius: '5px', background: '#f9f9f9', marginBottom: '10px' }}>
              <canvas 
                ref={canvasRef}

                onMouseUp={() => { if (canvasRef.current) setFirmaDataUrl(canvasRef.current.toDataURL()); }}
                onTouchEnd={() => { if (canvasRef.current) setFirmaDataUrl(canvasRef.current.toDataURL()); }}
              />
            </div>
            <button onClick={limpiarFirma} style={{ padding: '5px 10px', background: 'transparent', border: '1px solid #ccc', borderRadius: '3px', fontSize: '12px', cursor: 'pointer', marginBottom: '20px' }}>Limpiar Firma</button>

            <div style={{ display: 'flex', justifyContent: 'flex-end', gap: '10px' }}>
              <button onClick={() => setModalEntregaAbierto(false)} style={{ padding: '10px 20px', cursor: 'pointer', borderRadius: '5px', border: '1px solid #ccc' }}>Cancelar</button>
              <button onClick={confirmarEntregaMultiple} style={{ padding: '10px 20px', background: '#27ae60', color: 'white', border: 'none', borderRadius: '5px', cursor: 'pointer', fontWeight: 'bold' }}>Confirmar y Firmar</button>

            </div>
          </div>
        </div>
      )}
    </div>
  );
}
import { useState, useEffect } from 'react';
import { supabase } from '../supabaseClient';

export default function Clientes() {
  const [clientes, setClientes] = useState([]);
  const [mostrarFormulario, setMostrarFormulario] = useState(false);
  
  const [clienteActual, setClienteActual] = useState({
    id: null, nombre: '', ci_nit: '', telefono: '', direccion: '', email: ''
  });

  useEffect(() => {
    cargarClientes();
  }, []);

  const cargarClientes = async () => {
    const { data } = await supabase.from('clientes').select('*').order('nombre');
    if (data) setClientes(data);
  };

  const aMayusculas = (texto) => texto ? texto.toUpperCase() : '';

  const guardarCliente = async (e) => {
    e.preventDefault();
    
    const datosAGuardar = {
      nombre: aMayusculas(clienteActual.nombre),
      ci_nit: aMayusculas(clienteActual.ci_nit),
      telefono: clienteActual.telefono,
      direccion: aMayusculas(clienteActual.direccion),
      email: clienteActual.email ? clienteActual.email.toUpperCase() : ''
    };

    let error;
    if (clienteActual.id) {
      const { error: err } = await supabase.from('clientes').update(datosAGuardar).eq('id', clienteActual.id);
      error = err;
    } else {
      const { error: err } = await supabase.from('clientes').insert([datosAGuardar]);
      error = err;
    }

    if (error) {
      alert("Error: " + error.message);
    } else {
      cerrarFormulario();
      cargarClientes();
    }
  };

  const editarCliente = (cliente) => {
    setClienteActual(cliente);
    setMostrarFormulario(true);
  };

  const cerrarFormulario = () => {
    setClienteActual({ id: null, nombre: '', ci_nit: '', telefono: '', direccion: '', email: '' });
    setMostrarFormulario(false);
  };

  return (
    <div>
      {/* BARRA DE ACCIONES (Botón a la derecha) */}
      <div style={{ display: 'flex', justifyContent: 'flex-end', marginBottom: '20px' }}>
        {!mostrarFormulario && (
          <button onClick={() => setMostrarFormulario(true)} className="btn btn-primary">
            + Nuevo Cliente
          </button>
        )}
      </div>

      {/* FORMULARIO (Usando la clase .card de tu App.css) */}
      {mostrarFormulario && (
        <div className="card" style={{ marginBottom: '20px' }}>
          <h3 style={{ marginBottom: '20px' }}>{clienteActual.id ? 'Editar Cliente' : 'Nuevo Cliente'}</h3>
          
          <form onSubmit={guardarCliente}>
            <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '20px', marginBottom: '20px' }}>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Nombre Completo / Razón Social:</label>
                <input type="text" value={clienteActual.nombre} onChange={e => setClienteActual({...clienteActual, nombre: e.target.value})} required style={{ textTransform: 'uppercase' }} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>CI / NIT:</label>
                <input type="text" value={clienteActual.ci_nit} onChange={e => setClienteActual({...clienteActual, ci_nit: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Teléfono:</label>
                <input type="text" value={clienteActual.telefono} onChange={e => setClienteActual({...clienteActual, telefono: e.target.value})} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Email:</label>
                <input type="email" value={clienteActual.email} onChange={e => setClienteActual({...clienteActual, email: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
              <div style={{ gridColumn: 'span 2' }}>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Dirección:</label>
                <input type="text" value={clienteActual.direccion} onChange={e => setClienteActual({...clienteActual, direccion: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
            </div>

            <div style={{ display: 'flex', gap: '10px' }}>
              <button type="submit" className="btn btn-primary">
                {clienteActual.id ? 'Guardar Cambios' : 'Guardar Cliente'}
              </button>
              <button type="button" onClick={cerrarFormulario} className="btn btn-secondary">
                Cancelar
              </button>
            </div>
          </form>
        </div>
      )}

      {/* TABLA (Usando las clases globales de tu App.css) */}

      const draw = (e) => { if (!isDrawing) return; e.preventDefault(); const { x, y } = getCoordinates(e); ctx.lineTo(x, y); ctx.stroke(); };
      const stopDrawing = () => { setIsDrawing(false); };
      canvas.addEventListener('mousedown', startDrawing); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', stopDrawing); canvas.addEventListener('mouseleave', stopDrawing);
      canvas.addEventListener('touchstart', startDrawing, { passive: false }); canvas.addEventListener('touchmove', draw, { passive: false }); canvas.addEventListener('touchend', stopDrawing);

export default function MiTaller() {
  const [empleados, setEmpleados] = useState([]);
  const [empleadoSeleccionado, setEmpleadoSeleccionado] = useState('');
  const [ordenes, setOrdenes] = useState([]);
  const [cargoActual, setCargoActual] = useState(''); 
  
  // Estados para el modal de solicitudes
  const [modalMaterialAbierto, setModalMaterialAbierto] = useState(false);
  const [otSeleccionada, setOtSeleccionada] = useState(null);
  const [tipoSolicitud, setTipoSolicitud] = useState('REPUESTO');
  const [listaMaterialesOriginal, setListaMaterialesOriginal] = useState([]);
  const [listaMaterialesFiltrada, setListaMaterialesFiltrada] = useState([]);
  
  // Carrito
  const [carrito, setCarrito] = useState([]);
  const [materialSeleccionado, setMaterialSeleccionado] = useState('');
  const [cantidad, setCantidad] = useState(1);

  useEffect(() => {
    cargarEmpleados();
  }, []);

  useEffect(() => {
    if (empleadoSeleccionado) {
      const emp = empleados.find(e => e.id === empleadoSeleccionado);
      setCargoActual(emp ? emp.cargo : '');
      cargarMisOrdenes(empleadoSeleccionado);
    } else {
      setOrdenes([]);
      setCargoActual('');
    }
  }, [empleadoSeleccionado, empleados]);

  const cargarEmpleados = async () => {
    const { data } = await supabase.from('empleados').select('*');
    if (data) setEmpleados(data);
  };

   const cargarMisOrdenes = async (empId) => {
    // AQUÍ ESTÁ LA MAGIA: Solo traemos las OTs que NO estén finalizadas ni cerradas
    const { data: ots } = await supabase.from('ordenes_trabajo')
      .select('*')
      .eq('empleado_id', empId)
      .not('estado', 'in', '("FINALIZADO","CERRADO")') 
      .order('fecha_asignacion', { ascending: false });

    if (ots && ots.length > 0) {
      const ordenesCompletas = await Promise.all(ots.map(async (ot) => {
        const { data: veh } = await supabase.from('vehiculos').select('placa, marca, modelo').eq('id', ot.vehiculo_id).single();
        const { data: fases } = await supabase.from('ot_fases').select('*').eq('ot_id', ot.id);
        
        const { data: materiales, error: errMat } = await supabase.from('ot_materiales').select('*').eq('ot_id', ot.id);
        
        let materialesConNombres = [];
        if (materiales && !errMat) {
          materialesConNombres = await Promise.all(materiales.map(async (mat) => {
            let nombre = 'Material';
            if (mat.repuesto_id) {
              const { data: rep } = await supabase.from('recepcion_repuestos').select('nombre_repuesto').eq('id', mat.repuesto_id).single();
              if (rep) nombre = rep.nombre_repuesto;
            } else if (mat.insumo_id) {
              const { data: ins } = await supabase.from('insumos_taller').select('nombre').eq('id', mat.insumo_id).single();
              if (ins) nombre = ins.nombre;
            }
            return { ...mat, nombreMaterial: nombre };
          }));
        }

        return { ...ot, vehiculos: veh, ot_fases: fases || [], ot_materiales: materialesConNombres };
      }));
      setOrdenes(ordenesCompletas);
    } else {
      setOrdenes([]);
    }
  };

  const abrirModalMaterial = async (ot, tipo) => {
    setOtSeleccionada(ot);
    setTipoSolicitud(tipo);
    setModalMaterialAbierto(true);
    setCarrito([]); 
    setMaterialSeleccionado('');
    setCantidad(1);
    
    if (tipo === 'REPUESTO') {
      const { data: profData } = await supabase.from('proformas').select('id, numero_proforma').eq('vehiculo_id', ot.vehiculo_id).order('fecha', { ascending: false }).limit(1);
      
      if (profData && profData.length > 0) {
        const { data: repData } = await supabase.from('proforma_repuestos_solicitados').select('id, descripcion').eq('proforma_id', profData[0].id);
        const { data: llegadas } = await supabase.from('recepcion_repuestos').select('id, nombre_repuesto').eq('numero_proforma', profData[0].numero_proforma).neq('estado', 'RECHAZADO');
        
        if (repData) {
          const repuestosConEstado = repData.map(rep => {
            const llegada = llegadas?.find(l => l.nombre_repuesto.toUpperCase() === rep.descripcion.toUpperCase());
            return { 
              ...rep, 
              id_fisico: llegada ? llegada.id : null, 
              disponible: !!llegada 
            };
          });
          setListaMaterialesOriginal(repuestosConEstado);
          setListaMaterialesFiltrada(repuestosConEstado);
        }
      }
    } else {
      const { data: insData } = await supabase.from('insumos_taller').select('id, nombre, stock_actual').gt('stock_actual', 0);
      if (insData) {
        const insumosFormateados = insData.map(i => ({ ...i, disponible: true, id_fisico: i.id }));
        setListaMaterialesOriginal(insumosFormateados);
        setListaMaterialesFiltrada(insumosFormateados);
      }
    }
  };

  const agregarAlCarrito = () => {
    const materialInfo = listaMaterialesFiltrada.find(m => m.id === materialSeleccionado);
    if (!materialInfo) return alert("Seleccione un material");
    if (!materialInfo.disponible) return alert("⛔ Este repuesto aún no ha llegado al almacén.");

    const nombre = materialInfo.descripcion || materialInfo.nombre;
    const idParaGuardar = materialInfo.id_fisico; 

    setCarrito([...carrito, { id: idParaGuardar, nombre, cantidad, tipo: tipoSolicitud }]);
    
    const nuevaLista = listaMaterialesFiltrada.filter(m => m.id !== materialInfo.id);
    setListaMaterialesFiltrada(nuevaLista);

    setMaterialSeleccionado('');
    setCantidad(1);
  };

  const eliminarDelCarrito = (index) => {
    const nuevoCarrito = [...carrito];
    const itemEliminado = nuevoCarrito.splice(index, 1)[0];
    
    const materialOriginal = listaMaterialesOriginal.find(m => m.id_fisico === itemEliminado.id);
    if (materialOriginal) {
      setListaMaterialesFiltrada([...listaMaterialesFiltrada, materialOriginal]);
    }

    setCarrito(nuevoCarrito);
  };

  const enviarSolicitudMultiple = async () => {
    if (carrito.length === 0) return alert("La canasta está vacía");

    const solicitudes = carrito.map(item => ({
      ot_id: otSeleccionada.id,
      repuesto_id: item.tipo === 'REPUESTO' ? item.id : null,
      insumo_id: item.tipo === 'INSUMO' ? item.id : null,
      cantidad_solicitada: item.cantidad
    }));

    const { error } = await supabase.from('ot_materiales').insert(solicitudes);
    if (error) return alert("Error: " + error.message);

    alert(`¡Se enviaron ${carrito.length} solicitudes al Almacén!`);
    setModalMaterialAbierto(false);
    cargarMisOrdenes(empleadoSeleccionado);
  };

  const cambiarEstadoFase = async (faseId, nuevoEstado, otId) => {
    await supabase.from('ot_fases').update({ 
      estado: nuevoEstado,
      fecha_inicio: nuevoEstado === 'EN_PROCESO' ? new Date().toISOString() : null,
      fecha_fin: nuevoEstado === 'FINALIZADO' ? new Date().toISOString() : null
    }).eq('id', faseId);

    const { data: fasesActuales } = await supabase.from('ot_fases').select('estado').eq('ot_id', otId);
    if (fasesActuales) {
      const nuevoAvance = Math.round((fasesActuales.filter(f => f.estado === 'FINALIZADO').length / fasesActuales.length) * 100);
      const nuevoEstadoOT = nuevoAvance === 100 ? 'FINALIZADO' : 'EN_PROCESO';
      await supabase.from('ordenes_trabajo').update({ avance: nuevoAvance, estado: nuevoEstadoOT }).eq('id', otId);
    }
    cargarMisOrdenes(empleadoSeleccionado);
  };

  const agregarObservacion = async (otId, texto) => {
    await supabase.from('ordenes_trabajo').update({ observacion_chapista: texto.toUpperCase() }).eq('id', otId);
    alert("Observación guardada");
  };

  const styles = {
    card: { background: '#fff', borderRadius: '10px', boxShadow: '0 2px 8px rgba(0,0,0,0.04)', marginBottom: '12px', overflow: 'hidden', border: '1px solid #e1e8ed' },
    headerCard: { padding: '12px 15px', borderBottom: '1px solid #f0f0f0', display: 'flex', justifyContent: 'space-between', alignItems: 'center' },
    placa: { margin: 0, color: '#1e293b', fontSize: '16px', fontWeight: '800', letterSpacing: '0.5px' },
    modelo: { fontSize: '11px', color: '#64748b', marginTop: '0px' },
    avanceNum: { fontSize: '18px', fontWeight: '900', color: '#10b981' },
    faseRow: { display: 'flex', justifyContent: 'space-between', alignItems: 'center', padding: '6px 15px', borderBottom: '1px solid #f8fafc' },
    faseNombre: { fontSize: '12px', fontWeight: '600', color: '#334155' },
    btnIniciar: { padding: '4px 12px', background: '#eff6ff', color: '#3b82f6', border: 'none', borderRadius: '15px', fontSize: '11px', fontWeight: 'bold', cursor: 'pointer' },
    btnTerminar: { padding: '4px 12px', background: '#ecfdf5', color: '#10b981', border: 'none', borderRadius: '15px', fontSize: '11px', fontWeight: 'bold', cursor: 'pointer' },
    btnPedirRepuesto: { flex: 1, padding: '8px', background: '#fef3c7', color: '#d97706', border: 'none', borderRadius: '6px', fontSize: '12px', fontWeight: 'bold', cursor: 'pointer', display: 'flex', justifyContent: 'center', alignItems: 'center', gap: '3px' },
    btnPedirInsumo: { flex: 1, padding: '8px', background: '#f3e8ff', color: '#9333ea', border: 'none', borderRadius: '6px', fontSize: '12px', fontWeight: 'bold', cursor: 'pointer', display: 'flex', justifyContent: 'center', alignItems: 'center', gap: '3px' }
  };

  return (
    <div style={{ background: '#f8fafc', minHeight: '100vh', padding: '20px' }}>
      
      <div style={{ background: '#fff', padding: '15px 20px', borderRadius: '12px', marginBottom: '25px', boxShadow: '0 2px 8px rgba(0,0,0,0.04)', display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
        <span style={{ fontWeight: 'bold', color: '#475569', fontSize: '14px' }}>👤 Simulador de Login:</span>
        <select value={empleadoSeleccionado} onChange={e => setEmpleadoSeleccionado(e.target.value)} style={{ padding: '10px', borderRadius: '8px', border: '1px solid #cbd5e1', background: '#f8fafc', fontSize: '14px', outline: 'none', cursor: 'pointer' }}>
          <option value="">Seleccionar Trabajador...</option>
          {empleados.map(emp => <option key={emp.id} value={emp.id}>{emp.nombre} ({emp.cargo})</option>)}
        </select>
      </div>

      {!empleadoSeleccionado ? (
        <div style={{ textAlign: 'center', padding: '50px', color: '#94a3b8' }}>
          <h3>Selecciona un usuario para ver sus tareas</h3>
        </div>
      ) : ordenes.length === 0 ? (
        <div style={{ textAlign: 'center', padding: '50px', color: '#94a3b8' }}>
          <h3>No hay vehículos asignados a este trabajador</h3>
        </div>
      ) : (
        // AQUI ESTA EL CAMBIO CLAVE: GRID DE 2 COLUMNAS
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(auto-fill, minmax(350px, 1fr))', gap: '15px', maxWidth: '1200px', margin: '0 auto' }}>
          
          <div style={{ gridColumn: '1 / -1', background: cargoActual === 'MECANICO' ? '#e0f2fe' : '#fce7f3', padding: '10px 20px', borderRadius: '8px', marginBottom: '5px', color: cargoActual === 'MECANICO' ? '#0369a1' : '#be185d', fontWeight: 'bold', textAlign: 'center' }}>
            Flujo de Trabajo: {cargoActual}
          </div>

          {ordenes.map(ot => {
            const avance = ot.ot_fases.length === 0 ? 0 : Math.round((ot.ot_fases.filter(f => f.estado === 'FINALIZADO').length / ot.ot_fases.length) * 100);
            
            return (
              <div key={ot.id} style={styles.card}>
                
                <div style={styles.headerCard}>
                  <div>
                    <h3 style={styles.placa}>🚘 {ot.vehiculos?.placa}</h3>
                    <div style={styles.modelo}>{ot.vehiculos?.marca} {ot.vehiculos?.modelo}</div>
                  </div>
                  <div style={{ textAlign: 'right' }}>
                    <div style={styles.avanceNum}>{avance}%</div>
                    <div style={{ width: '80px', height: '6px', background: '#e2e8f0', borderRadius: '3px', marginTop: '5px', overflow: 'hidden' }}>
                      <div style={{ width: `${avance}%`, height: '100%', background: avance === 100 ? '#10b981' : '#3b82f6', transition: 'width 0.3s' }}></div>
                    </div>
                  </div>
                </div>

                <div>
                  {ot.ot_fases.map(fase => (
                    <div key={fase.id} style={styles.faseRow}>
                      <span style={{ ...styles.faseNombre, color: fase.estado === 'FINALIZADO' ? '#10b981' : '#334155' }}>
                        {fase.estado === 'FINALIZADO' ? '✅' : (fase.estado === 'EN_PROCESO' ? '🔄' : '⏳')} {fase.nombre_fase}
                      </span>
                      
                      {fase.estado === 'PENDIENTE' && (
                        <button onClick={() => cambiarEstadoFase(fase.id, 'EN_PROCESO', ot.id)} style={styles.btnIniciar}>▶ Iniciar</button>
                      )}
                      {fase.estado === 'EN_PROCESO' && (
                        <button onClick={() => cambiarEstadoFase(fase.id, 'FINALIZADO', ot.id)} style={styles.btnTerminar}>✔ Terminar</button>

                <div style={{ padding: '10px 15px', background: '#f8fafc' }}>

    const textoUpper = texto.toUpperCase();
    setProforma(prev => ({ ...prev, placa_buscada: textoUpper }));
    const veh = vehiculos.find(v => v.placa === textoUpper);
    if (veh) {
      setProforma(prev => ({ ...prev, vehiculo_id: veh.id, marca_modelo: `${veh.marca} ${veh.modelo}`, color_vehiculo: veh.color || '' }));
    } else {
      setProforma(prev => ({ ...prev, vehiculo_id: '', marca_modelo: '', color_vehiculo: '' }));
    }
  };

  const buscarCliente = (texto) => {
    const textoUpper = texto.toUpperCase();
    const cli = clientes.find(c => c.nombre === textoUpper);
    setProforma(prev => ({ ...prev, cliente_nombre: textoUpper, telefono_cliente: cli ? cli.telefono : '' }));
  };

  const agregarItem = () => setProforma({ ...proforma, items: [...proforma.items, { descripcion: '', cantidad: 1, precio_unitario: 0 }] });
  const actualizarItem = (index, campo, valor) => {
    const nuevosItems = [...proforma.items];
    nuevosItems[index][campo] = valor;
    setProforma({ ...proforma, items: nuevosItems });
  };

  const agregarRepuesto = () => setProforma({ ...proforma, repuestos: [...proforma.repuestos, { descripcion: '', cantidad: 1 }] });
  const actualizarRepuesto = (index, campo, valor) => {
    const nuevosRepuestos = [...proforma.repuestos];
    nuevosRepuestos[index][campo] = valor;
    setProforma({ ...proforma, repuestos: nuevosRepuestos });
  };

  const agregarTercero = () => setProforma({ ...proforma, terceros: [...proforma.terceros, { descripcion: '', precio_unitario: 0 }] });
  const actualizarTercero = (index, campo, valor) => {
    const nuevosTerceros = [...proforma.terceros];
    nuevosTerceros[index][campo] = valor;
    setProforma({ ...proforma, terceros: nuevosTerceros });
  };

  const calcularTotal = () => {
    const totalManoObra = proforma.items.reduce((total, item) => total + item.precio_unitario, 0);
    const totalTerceros = proforma.tiene_terceros ? proforma.terceros.reduce((total, ter) => total + ter.precio_unitario, 0) : 0;
    return totalManoObra + totalTerceros;
  };
  
  const totalEnLetras = (monto) => {
    if (monto === 0) return 'CERO BOLIVIANOS';
    return `${NumerosALetras(monto).toUpperCase()} BOLIVIANOS`;
  };

  const cargarDetalleProforma = async (prof) => {
    setProformaSeleccionada(prof);
    const { data: itemsData } = await supabase.from('proforma_items').select('*').eq('proforma_id', prof.id).not('descripcion', 'like', '[TERCEROS]%');
    const { data: repuestosData } = await supabase.from('proforma_repuestos_solicitados').select('*').eq('proforma_id', prof.id);
    const { data: tercerosData } = await supabase.from('proforma_items').select('*').eq('proforma_id', prof.id).like('descripcion', '[TERCEROS]%');

    setProforma({
      ...prof,
      items: itemsData || [], repuestos: repuestosData || [], terceros: tercerosData || [],
      tiene_terceros: tercerosData && tercerosData.length > 0,
      cliente_nombre: prof.clientes?.nombre || 'SIN NOMBRE', 
      placa_buscada: prof.vehiculos?.placa || 'SIN PLACA', 
      marca_modelo: `${prof.vehiculos?.marca || ''} ${prof.vehiculos?.modelo || ''}`, 
      color_vehiculo: prof.vehiculos?.color || '',
      telefono_cliente: prof.clientes?.telefono || ''
    });
  };

  const guardarProforma = async (e) => {
    e.preventDefault();
    
    const clienteEncontrado = clientes.find(c => c.nombre.toLowerCase() === proforma.cliente_nombre.toLowerCase());
    if (!clienteEncontrado) {
      alert("⛔ CLIENTE NO REGISTRADO. Use el botón '+ NUEVO' para registrarlo.");
      return; 
    }

    const vehiculoEncontrado = vehiculos.find(v => v.placa === proforma.placa_buscada);
    if (!vehiculoEncontrado) {
      alert("⛔ VEHÍCULO NO REGISTRADO. Use el botón '+ NUEVO' para registrarlo.");
      return; 
    }

    const { data: profGuardada, error } = await supabase.from('proformas').insert([{ 
        vehiculo_id: vehiculoEncontrado.id, 
        cliente_id: clienteEncontrado.id, 
        total: calcularTotal(),
        numero_proforma: proforma.numero_proforma, 
        seguro: proforma.tiene_seguro ? proforma.seguro : 'SIN SEGURO',
        es_rc: proforma.es_rc, 
        placa_asegurado: proforma.placa_asegurado
      }]).select();

    if (error) return alert("Error: " + error.message);

    const itemsValidos = proforma.items.filter(item => item.descripcion.trim() !== '');
    if (itemsValidos.length > 0) {
      await supabase.from('proforma_items').insert(itemsValidos.map(item => ({ proforma_id: profGuardada[0].id, descripcion: item.descripcion, cantidad: 1, precio_unitario: item.precio_unitario })));
    }

    const repuestosValidos = proforma.repuestos.filter(rep => rep.descripcion.trim() !== '');
    if (repuestosValidos.length > 0) {
      await supabase.from('proforma_repuestos_solicitados').insert(repuestosValidos.map(rep => ({ proforma_id: profGuardada[0].id, descripcion: rep.descripcion, cantidad: 1 })));
    }

    if (proforma.tiene_terceros) {
      const tercerosValidos = proforma.terceros.filter(ter => ter.descripcion.trim() !== '');
      if (tercerosValidos.length > 0) {
        await supabase.from('proforma_items').insert(tercerosValidos.map(ter => ({ proforma_id: profGuardada[0].id, descripcion: `[TERCEROS] ${ter.descripcion}`, cantidad: 1, precio_unitario: ter.precio_unitario })));
      }
    }

    alert("¡Proforma guardada exitosamente!");
    setMostrarFormulario(false);
    cargarDatos();
    setProforma(estadoInicial); 
  };

  return (
        <div>
      <div className="print-hide">
        {/* BARRA DE ACCIONES LIMPIA */}
        <div style={{ display: 'flex', justifyContent: 'space-between', marginBottom: '20px', alignItems: 'center' }}>
          <button onClick={() => { setProforma(estadoInicial); setMostrarFormulario(true); setProformaSeleccionada(null); }} style={{ padding: '10px 20px', background: '#1e4b8f', color: 'white', border: 'none', borderRadius: '5px', cursor: 'pointer', fontSize: '16px', fontWeight: 'bold' }}>
            + Crear Nueva Proforma
          </button>
          <input type="text" placeholder="🔍 Buscar por Placa, Cliente o N° Proforma..." value={busqueda} onChange={(e) => setBusqueda(e.target.value)} style={{ padding: '10px', width: '350px', borderRadius: '5px', border: '1px solid #ccc', fontSize: '14px' }} />
        </div>

        <h3>Historial de Proformas</h3>
        <table style={{ width: '100%', background: 'white', borderCollapse: 'collapse', textAlign: 'left' }}>
          <thead>
            <tr style={{ background: '#eee' }}>
              <th style={{ padding: '10px' }}>N° Proforma</th>
              <th style={{ padding: '10px' }}>Cliente</th>
              <th style={{ padding: '10px' }}>Placa</th>
              <th style={{ padding: '10px' }}>Total (Bs)</th>
              <th style={{ padding: '10px' }}>Acciones</th>
            </tr>
          </thead>
          <tbody>
            {proformasFiltradas.map(p => (
              <tr key={p.id} style={{ borderBottom: '1px solid #ddd' }}>
                <td style={{ padding: '10px' }}>{p.numero_proforma}</td>
                <td style={{ padding: '10px' }}>{p.clientes?.nombre || 'Cliente Ocasional'}</td>
                <td style={{ padding: '10px' }}>{p.vehiculos?.placa}</td>
                <td style={{ padding: '10px' }}>{p.total}</td>
                <td style={{ padding: '10px' }}>
                  <button onClick={() => cargarDetalleProforma(p)} style={{ padding: '5px 10px', background: '#3498db', color: 'white', border: 'none', borderRadius: '3px', cursor: 'pointer' }}>Ver / Imprimir</button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {(mostrarFormulario || proformaSeleccionada) && (
        <div className="modal-overlay">
          <div className="modal-content">
            <div className="paper-container uppercase-paper" style={{ boxShadow: 'none', margin: 0 }}>
              
              <div className="print-hide" style={{ display: 'flex', justifyContent: 'flex-end', gap: '10px', marginBottom: '20px' }}>
                <button onClick={() => { setMostrarFormulario(false); setProformaSeleccionada(null); setProforma(estadoInicial); }} style={{ padding: '10px 20px', cursor: 'pointer' }}>Cerrar</button>
                <button onClick={() => window.print()} style={{ padding: '10px 20px', background: '#27ae60', color: 'white', border: 'none', borderRadius: '5px', cursor: 'pointer' }}>🖨️ Imprimir / PDF</button>
                {mostrarFormulario && <button onClick={guardarProforma} style={{ padding: '10px 20px', background: '#1e4b8f', color: 'white', border: 'none', borderRadius: '5px', cursor: 'pointer' }}>Guardar</button>}
              </div>

              {/* CABECERA */}
              <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start', borderBottom: '3px solid #1e4b8f', paddingBottom: '10px', marginBottom: '15px' }}>
                <div style={{ display: 'flex', flexDirection: 'column', width: '40%', alignItems: 'center' }}>
                  <img src="/logo-mecanicorp.png" alt="Logo" style={{ height: '110px', objectFit: 'contain' }} />
                  <div style={{ fontSize: '9px', color: '#1e4b8f', textAlign: 'center', marginTop: '5px', lineHeight: '1.3', fontWeight: 'bold' }}>
                    RADIAL 27 BARRIO ENDE CALLE 3 / PARALELA AV. BANZER Z/NORTE<br/>
                    TELF. OF.: 77605007 - 77603006<br/>
                    SANTA CRUZ - BOLIVIA
                  </div>
                </div>
                <div style={{ textAlign: 'right', width: '60%' }}>
                  <div style={{ fontSize: '12px', fontWeight: 'bold', color: '#1e4b8f', marginBottom: '5px' }}>CHAPERIO - PINTURA - RAMPLA<br/>MECÁNICA - INYECCIÓN ELECTRÓNICA</div>
                  <div style={{ border: '2px solid #1e4b8f', color: '#1e4b8f', fontSize: '24px', fontWeight: 'bold', padding: '5px 20px', borderRadius: '5px', display: 'inline-block', marginBottom: '5px' }}>PRO-FORMA</div>
                  <div style={{ fontSize: '11px', fontWeight: 'bold' }}>STEVEN ARAS - 70070044 - 70000570</div>
                  <div style={{ fontSize: '16px', fontWeight: 'bold', marginTop: '5px' }}>
                    <strong>Nº </strong> 
                    {mostrarFormulario ? (
                      <input type="text" value={proforma.numero_proforma} onChange={e => setProforma({...proforma, numero_proforma: e.target.value})} style={{ width: '100px', fontWeight: 'bold', fontSize: '16px', color: '#e74c3c' }} />
                    ) : <span style={{ color: '#e74c3c' }}>{proformaSeleccionada?.numero_proforma}</span>}
                  </div>
                </div>
              </div>

              {/* DATOS DEL CLIENTE Y VEHICULO */}
              <div style={{ display: 'flex', flexDirection: 'column', gap: '10px', marginBottom: '20px', fontSize: '13px' }}>
                <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-end' }}>
                  <div style={{ width: '65%', display: 'flex', alignItems: 'flex-end', gap: '10px' }}>
                    <strong style={{ marginRight: '5px', whiteSpace: 'nowrap' }}>CLIENTE: </strong>
                    {mostrarFormulario ? (
                      <>
                        <input type="text" list="lista-clientes" value={proforma.cliente_nombre} onChange={e => buscarCliente(e.target.value)} style={{ width: '100%', borderBottom: '1px solid #000' }} />
                        <datalist id="lista-clientes">{clientes.map(c => <option key={c.id} value={c.nombre} />)}</datalist>
                        <button type="button" onClick={() => setModalClienteAbierto(true)} style={{ fontSize: '11px', background: '#e74c3c', color: 'white', padding: '3px 8px', border: 'none', borderRadius: '3px', cursor: 'pointer', whiteSpace: 'nowrap', fontWeight: 'bold' }}>+ NUEVO</button>
                      </>
                    ) : proformaSeleccionada?.clientes?.nombre}
                  </div>
                  <div style={{ width: '30%' }}>
                    <strong>TELÉFONO: </strong> {mostrarFormulario ? proforma.telefono_cliente : proformaSeleccionada?.clientes?.telefono}
                  </div>
                </div>

                <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-end' }}>
                                   <div style={{ width: '30%', display: 'flex', alignItems: 'flex-end', gap: '10px' }}>
                    <strong style={{ marginRight: '5px', whiteSpace: 'nowrap' }}>VEHÍCULO: </strong>
                    {mostrarFormulario ? (
                      <>
                        <input type="text" list="lista-placas" placeholder="PLACA" value={proforma.placa_buscada} onChange={(e) => buscarPorPlaca(e.target.value)} style={{ width: '80px', borderBottom: '1px solid #000' }} />
                        <datalist id="lista-placas">{vehiculos.map(v => <option key={v.id} value={v.placa} />)}</datalist>
                        {/* BOTÓN SIEMPRE VISIBLE */}
                        <button type="button" onClick={() => setModalVehiculoAbierto(true)} style={{ position: 'relative', zIndex: 10, fontSize: '11px', background: '#e74c3c', color: 'white', padding: '3px 8px', border: 'none', borderRadius: '3px', cursor: 'pointer', whiteSpace: 'nowrap', fontWeight: 'bold' }}>+ NUEVO</button>
                      </>
                    ) : proformaSeleccionada?.vehiculos?.placa}
                  </div>
                  <div style={{ width: '40%' }}>
                    <strong>MODELO: </strong>
                    {mostrarFormulario ? <span>{proforma.marca_modelo}</span> : `${proformaSeleccionada?.vehiculos?.marca || ''} ${proformaSeleccionada?.vehiculos?.modelo || ''}`}
                  </div>
                  <div style={{ width: '30%' }}>
                    <strong>COLOR: </strong>
                    {mostrarFormulario ? <span>{proforma.color_vehiculo}</span> : proformaSeleccionada?.vehiculos?.color}
                  </div>
                </div>

                <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginTop: '5px' }}>
                  <div style={{ width: '65%', display: 'flex', gap: '10px', alignItems: 'center' }}>
                    <label style={{ display: 'flex', alignItems: 'center', gap: '5px', cursor: 'pointer' }}>
                      <input type="checkbox" checked={mostrarFormulario ? proforma.tiene_seguro : proformaSeleccionada?.seguro !== 'SIN SEGURO'} onChange={e => setProforma({...proforma, tiene_seguro: e.target.checked})} disabled={!mostrarFormulario} />
                      <strong>SEGURO:</strong>
                    </label>
                    {(mostrarFormulario ? proforma.tiene_seguro : proformaSeleccionada?.seguro !== 'SIN SEGURO') && (
                      <>
                        <input type="text" placeholder="ASEGURADORA" value={mostrarFormulario ? proforma.seguro : proformaSeleccionada?.seguro} onChange={e => setProforma({...proforma, seguro: e.target.value})} style={{ width: '150px', borderBottom: '1px solid #000' }} disabled={!mostrarFormulario} />
                        <label style={{ display: 'flex', alignItems: 'center', gap: '5px', cursor: 'pointer', marginLeft: '10px' }}>
                          <input type="checkbox" checked={mostrarFormulario ? proforma.es_rc : proformaSeleccionada?.es_rc} onChange={e => setProforma({...proforma, es_rc: e.target.checked})} disabled={!mostrarFormulario} />
                          <strong>RC DE</strong>
                        </label>
                        {(mostrarFormulario ? proforma.es_rc : proformaSeleccionada?.es_rc) && <input type="text" placeholder="PLACA ASEG." value={mostrarFormulario ? proforma.placa_asegurado : proformaSeleccionada?.placa_asegurado} onChange={e => setProforma({...proforma, placa_asegurado: e.target.value.toUpperCase()})} style={{ width: '80px', borderBottom: '1px solid #000' }} disabled={!mostrarFormulario} />}
                      </>
                    )}
                  </div>
                  <div style={{ width: '30%' }}>
                    <strong>FECHA: </strong> {mostrarFormulario ? proforma.fecha : new Date(proformaSeleccionada?.fecha).toLocaleDateString('es-BO')}
                  </div>
                </div>
              </div>

              {/* TABLA DE MANO DE OBRA / ITEMS */}
              <h4 style={{ marginBottom: '5px', textAlign: 'center', fontSize: '14px' }}>DESCRIPCIÓN</h4>
              <table style={{ width: '100%', borderCollapse: 'collapse', marginBottom: '10px' }}>
                <thead>
                  <tr>
                    <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'center', color: '#1e4b8f', width: '50px' }}>ITEM</th>
                    <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left', color: '#1e4b8f', width: '100%' }}>DETALLE</th>
                    <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'center', color: '#1e4b8f', width: '120px' }}>TOTAL</th>
                  </tr>
                </thead>
                <tbody>
                  {mostrarFormulario ? proforma.items.map((item, index) => (
                      <tr key={index}>
                        <td style={{ border: '1px solid #1e4b8f', textAlign: 'center', fontWeight: 'bold', width: '50px' }}>{index + 1}</td>
                        <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left' }}><input type="text" placeholder="EJ. CAMBIO Y BALANCEO..." value={item.descripcion} onChange={e => actualizarItem(index, 'descripcion', e.target.value)} style={{ width: '100%', padding: '3px', border: 'none', background: 'transparent', textAlign: 'left' }} /></td>
                        <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'right', width: '120px' }}>
                          <input type="number" placeholder="0" value={item.precio_unitario === 0 ? '' : item.precio_unitario} onChange={e => actualizarItem(index, 'precio_unitario', parseFloat(e.target.value) || 0)} style={{ width: '100%', textAlign: 'right', padding: '3px', border: 'none', background: 'transparent' }} />
                        </td>
                      </tr>
                    )) : proforma.items?.map((item, index) => (
                      <tr key={index}>
                        <td style={{ border: '1px solid #1e4b8f', textAlign: 'center', fontWeight: 'bold', width: '50px' }}>{index + 1}</td>
                        <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left' }}>{item.descripcion}</td>
                        <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'right', width: '120px' }}>{item.precio_unitario}</td>
                      </tr>
                    ))}
                </tbody>
              </table>
              {mostrarFormulario && <button className="print-hide" onClick={agregarItem} style={{ padding: '5px 10px', cursor: 'pointer', marginBottom: '20px' }}>+ AGREGAR LÍNEA</button>}

              {/* TABLA DE SOLICITUD DE REPUESTOS */}
              <h4 style={{ marginBottom: '5px', fontSize: '14px' }}>* SOLICITUD DE REPUESTOS</h4>
              <table style={{ width: '100%', borderCollapse: 'collapse', marginBottom: '10px' }}>
                <thead>
                  <tr>
                    <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'center', color: '#1e4b8f', width: '50px' }}>ITEM</th>
                    <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left', color: '#1e4b8f', width: '100%' }}>DESCRIPCIÓN DEL REPUESTO</th>
                  </tr>
                </thead>
                <tbody>
                  {mostrarFormulario ? proforma.repuestos.map((rep, index) => (
                      <tr key={index}>
                        <td style={{ border: '1px solid #1e4b8f', textAlign: 'center', fontWeight: 'bold', width: '50px' }}>{index + 1}</td>
                        <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left' }}><input type="text" placeholder="EJ. FAROL DELANTERO..." value={rep.descripcion} onChange={e => actualizarRepuesto(index, 'descripcion', e.target.value)} style={{ width: '100%', padding: '3px', border: 'none', background: 'transparent', textAlign: 'left' }} /></td>
                      </tr>
                    )) : proforma.repuestos?.map((rep, index) => (
                      <tr key={index}>
                        <td style={{ border: '1px solid #1e4b8f', textAlign: 'center', fontWeight: 'bold', width: '50px' }}>{index + 1}</td>
                        <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left' }}>{rep.descripcion}</td>
                      </tr>
                    ))}
                </tbody>
              </table>
              {mostrarFormulario && <button className="print-hide" onClick={agregarRepuesto} style={{ padding: '5px 10px', cursor: 'pointer', marginBottom: '20px' }}>+ AGREGAR REPUESTO</button>}

              {/* CHECKBOX Y TABLA DE TRABAJOS DE TERCEROS */}
              <div style={{ marginTop: '10px', marginBottom: '10px' }}>
                <label style={{ display: 'flex', alignItems: 'center', gap: '5px', cursor: 'pointer', fontSize: '13px', fontWeight: 'bold' }}>
                  <input type="checkbox" checked={mostrarFormulario ? proforma.tiene_terceros : proformaSeleccionada?.tiene_terceros} onChange={e => setProforma({...proforma, tiene_terceros: e.target.checked})} disabled={!mostrarFormulario} />
                  INCLUYE TRABAJOS DE TERCEROS
                </label>
              </div>

              {(mostrarFormulario ? proforma.tiene_terceros : proformaSeleccionada?.tiene_terceros) && (
                <>
                  <h4 style={{ marginBottom: '5px', fontSize: '14px' }}>TRABAJOS DE TERCEROS</h4>
                  <table style={{ width: '100%', borderCollapse: 'collapse', marginBottom: '10px' }}>
                    <thead>
                      <tr>
                        <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'center', color: '#1e4b8f', width: '50px' }}>ITEM</th>
                        <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left', color: '#1e4b8f', width: '100%' }}>SERVICIO DE TERCEROS</th>
                        <th style={{ background: '#e0e7ff', border: '1px solid #1e4b8f', padding: '5px', textAlign: 'center', color: '#1e4b8f', width: '120px' }}>TOTAL</th>
                      </tr>
                    </thead>
                    <tbody>
                      {mostrarFormulario ? proforma.terceros.map((ter, index) => (
                          <tr key={index}>
                            <td style={{ border: '1px solid #1e4b8f', textAlign: 'center', fontWeight: 'bold', width: '50px' }}>{index + 1}</td>
                            <td style={{ border: '1px solid #1e4b8f', padding: '5px', textAlign: 'left' }}><input type="text" placeholder="EJ. REPARACIÓN DE AIRE..." value={ter.descripcion} onChange={e => actualizarTercero(index, 'descripcion', e.target.value)} style={{ width: '100%', padding: '3px', border: 'none', background: 'transparent', textAlign: 'left' }} /></td>

import { supabase } from '../supabaseClient';
import ModalCliente from '../components/ModalCliente';
import ModalVehiculo from '../components/ModalVehiculo';

export default function Recepcion() {
  const [vista, setVista] = useState('LISTA'); 
  const [recepcionesGuardadas, setRecepcionesGuardadas] = useState([]);
  const [busquedaHistorial, setBusquedaHistorial] = useState('');
  
  // Datos para detalle
  const [recepcionSeleccionada, setRecepcionSeleccionada] = useState(null);
  const [checklistDetalle, setChecklistDetalle] = useState([]);
  const [danosDetalle, setDanosDetalle] = useState([]);
  const [fotosDetalle, setFotosDetalle] = useState([]);
  const [telefonoDetalle, setTelefonoDetalle] = useState('');
  
  // Modales de Fotos y Detalle
  const [modalFotosAbierto, setModalFotosAbierto] = useState(false);
  const [fotoAmpliada, setFotoAmpliada] = useState(null);
  
  // ESTADOS PARA EL ZOOM Y PAN
  const [zoom, setZoom] = useState(1);
  const [pan, setPan] = useState({ x: 0, y: 0 });
  const [isDragging, setIsDragging] = useState(false);
  const [dragStart, setDragStart] = useState({ x: 0, y: 0 });
  // Estados para la ventana personalizada y el botón de guardar
  const [alerta, setAlerta] = useState({ visible: false, mensaje: '' });
  const [guardando, setGuardando] = useState(false); // <-- Nuevo
  // Estados del Formulario
  const [vehiculos, setVehiculos] = useState([]);
  const [clientes, setClientes] = useState([]);
  const [placaBusqueda, setPlacaBusqueda] = useState('');
  const [vehiculoSeleccionado, setVehiculoSeleccionado] = useState(null);

  const [telefono, setTelefono] = useState('');

  const [combustible, setCombustible] = useState('');
  const [obsGenerales, setObsGenerales] = useState('');
  const [tieneSeguro, setTieneSeguro] = useState(false);
  const [aseguradora, setAseguradora] = useState('');
  
  const [checklist, setChecklist] = useState([
    { item: 'Luces principales', estado: 'SI' }, { item: 'Luces medias / Stop', estado: 'SI' },
    { item: 'Limpiaparabrisas', estado: 'SI' }, { item: 'Radio / Pantalla', estado: 'SI' },
    { item: 'Espejo LH', estado: 'SI' }, { item: 'Espejo RH', estado: 'SI' },
    { item: 'Vidrios laterales', estado: 'SI' }, { item: 'Llanta de Auxilio', estado: 'SI' },
    { item: 'Gato Hidráulico', estado: 'SI' }, { item: 'Llave de Ruedas', estado: 'SI' },
    { item: 'Triángulos', estado: 'SI' }, { item: 'Botiquín', estado: 'SI' },
    { item: 'Tapa Aros', estado: 'SI' }, { item: 'Emblemas', estado: 'SI' },
    { item: 'Tapa de Gasolina', estado: 'SI' },

    nuevasFotos[index].angulo = nuevoAngulo;
    setFotos(nuevasFotos);

    nuevasFotos.splice(index, 1);
    setFotos(nuevasFotos);

    setKilometraje(''); setCombustible(''); setObsGenerales(''); setDanos([]); setFotos([]);
    setTieneSeguro(false); setAseguradora('');
    setChecklist(checklist.map(c => ({ ...c, estado: 'SI' })));
  };

    const guardarRecepcion = async (e) => {
    e.preventDefault();
    if (!vehiculoSeleccionado) return alert("⛔ VEHÍCULO NO REGISTRADO. Regístrelo primero.");

    const { data: recGuardada, error } = await supabase.from('recepciones').insert([{

    })));

        recepcion_id: idRecepcion, tipo_dano: d.tipo, posicion_x: d.x, posicion_y: d.y, observacion: d.observacion
      })));
    }

    for (const foto of fotos) {
      const fileExt = foto.file.name.split('.').pop();

          recepcion_id: idRecepcion, url_foto: urlData.publicUrl, angulo: foto.angulo
        }]);

     setAlerta({ visible: true, mensaje: '¡Recepción guardada exitosamente!' });

    setZoom(prev => Math.min(Math.max(1, prev + delta), 5));
  };

  const handleMouseDown = (e) => {
    if (zoom > 1) {
      setIsDragging(true);
      setDragStart({ x: e.clientX - pan.x, y: e.clientY - pan.y });
    }
  };

    if (isDragging && zoom > 1) {
      setPan({ x: e.clientX - dragStart.x, y: e.clientY - dragStart.y });
    }

    setIsDragging(false);

              {esDetalle ? <span style={{ flex: 1, borderBottom: '1px solid #000' }}>{datosMostrar.telefono}</span> :
                <input type="text" value={telefono} onChange={e => setTelefono(e.target.value)} style={{ flex: 1, borderBottom: '1px solid #000', border: 'none', background: 'transparent' }} />

            </div>

            <div style={{ display: 'flex', alignItems: 'flex-end', marginBottom: '10px' }}>
              <strong style={{ width: '80px' }}>Placa:</strong>
              {esDetalle ? <span style={{ flex: 1, borderBottom: '1px solid #000', fontWeight: 'bold' }}>{datosMostrar.placa}</span> :

}import { useState, useEffect } from 'react';
import { supabase } from '../supabaseClient';

export default function Vehiculos() {
  const [vehiculos, setVehiculos] = useState([]);
  const [mostrarFormulario, setMostrarFormulario] = useState(false);
  
  // Estado para el vehículo que estamos creando o editando
  const [vehiculoActual, setVehiculoActual] = useState({
    id: null,
    placa: '', tipo: '', marca: '', modelo: '', anio: '', color: ''
  });

  useEffect(() => {
    cargarVehiculos();
  }, []);

  const cargarVehiculos = async () => {
    const { data } = await supabase.from('vehiculos').select('*').order('fecha_registro', { ascending: false });
    if (data) setVehiculos(data);
  };

  const aMayusculas = (texto) => {
    return texto ? texto.toUpperCase() : '';
  };

  const guardarVehiculo = async (e) => {
    e.preventDefault();
    
    if (!vehiculoActual.id) {
      const existe = vehiculos.some(v => v.placa.toUpperCase() === vehiculoActual.placa.toUpperCase());
      if (existe) {
        alert("Esta placa ya está registrada en el sistema.");
        return;
      }
    }

    const datosAGuardar = {
      placa: aMayusculas(vehiculoActual.placa),
      tipo: aMayusculas(vehiculoActual.tipo),
      marca: aMayusculas(vehiculoActual.marca),
      modelo: aMayusculas(vehiculoActual.modelo),
      anio: vehiculoActual.anio ? parseInt(vehiculoActual.anio) : null,
      color: aMayusculas(vehiculoActual.color)
    };

    let error;
    if (vehiculoActual.id) {
      const { error: err } = await supabase.from('vehiculos').update(datosAGuardar).eq('id', vehiculoActual.id);
      error = err;
    } else {
      const { error: err } = await supabase.from('vehiculos').insert([datosAGuardar]);
      error = err;
    }

    if (error) {
      alert("Error al guardar vehículo: " + error.message);
    } else {
      cerrarFormulario();
      cargarVehiculos();
    }
  };

  const editarVehiculo = (vehiculo) => {
    setVehiculoActual(vehiculo);
    setMostrarFormulario(true);
  };

  const cerrarFormulario = () => {
    setVehiculoActual({ id: null, placa: '', tipo: '', marca: '', modelo: '', anio: '', color: '' });
    setMostrarFormulario(false);
  };

  return (
    <div>
      {/* BARRA DE ACCIONES (Botón a la derecha) */}
      <div style={{ display: 'flex', justifyContent: 'flex-end', marginBottom: '20px' }}>
        {!mostrarFormulario && (
          <button onClick={() => setMostrarFormulario(true)} className="btn btn-primary">
            + Nuevo Vehículo
          </button>
        )}
      </div>

      {/* FORMULARIO */}
      {mostrarFormulario && (
        <div className="card" style={{ marginBottom: '20px' }}>
          <h3 style={{ marginBottom: '20px' }}>{vehiculoActual.id ? 'Editar Vehículo' : 'Nuevo Vehículo'}</h3>
          
          <form onSubmit={guardarVehiculo}>
            <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '20px', marginBottom: '20px' }}>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Placa (Única):</label>
                <input type="text" value={vehiculoActual.placa} onChange={e => setVehiculoActual({...vehiculoActual, placa: e.target.value})} required style={{ textTransform: 'uppercase' }} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Tipo:</label>
                <input type="text" value={vehiculoActual.tipo} onChange={e => setVehiculoActual({...vehiculoActual, tipo: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Marca:</label>
                <input type="text" value={vehiculoActual.marca} onChange={e => setVehiculoActual({...vehiculoActual, marca: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Modelo:</label>
                <input type="text" value={vehiculoActual.modelo} onChange={e => setVehiculoActual({...vehiculoActual, modelo: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Año:</label>
                <input type="number" value={vehiculoActual.anio} onChange={e => setVehiculoActual({...vehiculoActual, anio: e.target.value})} />
              </div>
              <div>
                <label style={{ display: 'block', marginBottom: '5px', fontSize: '14px' }}>Color:</label>
                <input type="text" value={vehiculoActual.color} onChange={e => setVehiculoActual({...vehiculoActual, color: e.target.value})} style={{ textTransform: 'uppercase' }} />
              </div>
            </div>

            <div style={{ display: 'flex', gap: '10px' }}>
              <button type="submit" className="btn btn-primary">
                {vehiculoActual.id ? 'Guardar Cambios' : 'Guardar Vehículo'}
              </button>
              <button type="button" onClick={cerrarFormulario} className="btn btn-secondary">
                Cancelar
              </button>
            </div>
          </form>
        </div>
      )}

      {/* TABLA */}

export default function ModalVehiculo({ onClose, onVehiculoAgregado }) {
  const [vehiculo, setVehiculo] = useState({ placa: '', tipo: '', marca: '', modelo: '', anio: '', color: '', chasis: '' });
  const [cargando, setCargando] = useState(false);

  const guardar = async (e) => {
    e.preventDefault();
    setCargando(true);
    
    // Verificamos si la placa ya existe
    const { data: existe } = await supabase.from('vehiculos').select('id').ilike('placa', vehiculo.placa).limit(1);
    if (existe && existe.length > 0) {
      setCargando(false);
      return alert("¡Error! Esta placa ya está registrada.");
    }

    const { data, error } = await supabase.from('vehiculos').insert([{
      placa: vehiculo.placa.toUpperCase(),
      tipo: vehiculo.tipo.toUpperCase(),
      marca: vehiculo.marca.toUpperCase(),
      modelo: vehiculo.modelo.toUpperCase(),
      anio: parseInt(vehiculo.anio) || null,
      color: vehiculo.color.toUpperCase(),
      chasis: vehiculo.chasis.toUpperCase()
    }]).select();

    setCargando(false);
    if (error) return alert("Error al guardar vehículo: " + error.message);
    
    alert("¡Vehículo registrado!");
    onVehiculoAgregado(data[0]); // Le avisa a la Proforma
    onClose(); 
  };

  return (
    <div style={{ position: 'fixed', top: 0, left: 0, width: '100vw', height: '100vh', background: 'rgba(0,0,0,0.6)', display: 'flex', justifyContent: 'center', alignItems: 'center', zIndex: 2000 }}>
      <div style={{ background: 'white', padding: '20px', borderRadius: '8px', width: '450px', maxWidth: '90%' }}>
        <h3 style={{ marginBottom: '15px' }}>Registrar Nuevo Vehículo</h3>
        <form onSubmit={guardar}>
          <input type="text" placeholder="Placa" value={vehiculo.placa} onChange={e => setVehiculo({...vehiculo, placa: e.target.value})} required style={{ width: '100%', padding: '8px', marginBottom: '10px', textTransform: 'uppercase', fontWeight: 'bold' }} />
          
          <div style={{ display: 'flex', gap: '10px', marginBottom: '10px' }}>
            <input type="text" placeholder="Marca" value={vehiculo.marca} onChange={e => setVehiculo({...vehiculo, marca: e.target.value})} style={{ flex: 1, padding: '8px', textTransform: 'uppercase' }} />
            <input type="text" placeholder="Modelo" value={vehiculo.modelo} onChange={e => setVehiculo({...vehiculo, modelo: e.target.value})} style={{ flex: 1, padding: '8px', textTransform: 'uppercase' }} />
          </div>
          
          <div style={{ display: 'flex', gap: '10px', marginBottom: '10px' }}>
            <input type="text" placeholder="Tipo (Auto, Camioneta)" value={vehiculo.tipo} onChange={e => setVehiculo({...vehiculo, tipo: e.target.value})} style={{ flex: 1, padding: '8px', textTransform: 'uppercase' }} />
            <input type="number" placeholder="Año" value={vehiculo.anio} onChange={e => setVehiculo({...vehiculo, anio: e.target.value})} style={{ flex: 1, padding: '8px' }} />
          </div>

          <input type="text" placeholder="Color" value={vehiculo.color} onChange={e => setVehiculo({...vehiculo, color: e.target.value})} style={{ width: '100%', padding: '8px', marginBottom: '10px', textTransform: 'uppercase' }} />
          <input type="text" placeholder="N° Chasis" value={vehiculo.chasis} onChange={e => setVehiculo({...vehiculo, chasis: e.target.value})} style={{ width: '100%', padding: '8px', marginBottom: '15px', textTransform: 'uppercase' }} />

            </button>
          </div>
        </form>
      </div>
    </div>
  );
